# Synthetic Dawn - Resource_Consolidation
resource_consolidation_quest = {
	length = 1
	icon = "GFX_edict_type_policy"
	resources = {
		category = edicts
	}
	potential = {
		AND = {
			is_ai = no
			years_passed < @yearsBeforeDeadline
			NOT = {
				OR = {
					has_country_flag = decision_resource_consolidation
					has_origin = origin_machine
					#has_country_flag = decision_origins_resource_consolidation #From xhk_decision_origins
				}
			}
		}
	}

	allow = {
		custom_tooltip = {
			fail_text = "requires_home_planet"
			is_nomadic = no
		}
	}
	
	effect = {
		if = {
			limit = { any_owned_species = { has_trait = trait_organic}}
			country_event = { id = warning_event.01 }
		}
		hidden_effect = {
			set_country_flag = decision_resource_consolidation
			country_event = { id = timeline_MMO.03 days = 4}

			capital_scope = {
				change_pc = pc_machine
				add_deposit = d_machine_prototype_pc_machine
				add_deposit = d_machine_ongoing_construction
				add_deposit = d_machine_ongoing_construction
				if = { #Assimilators
					limit = { owner = { has_civic = civic_machine_assimilator } }
					add_deposit = d_organic_slurry
				}

				add_zone = {
					district = district_nexus_1
					zone = zone_energy_nexus
				}
				add_zone = {
					district = district_nexus_2
					zone = zone_minerals_nexus
				}
			}
			owner = {
				capital_scope = {
					add_deposit = d_machine_minerals
					solar_system = {
						every_system_planet = {
							limit = {
								colonizable_planet = no
								is_star = no
							}
							clear_deposits = yes
						}
						every_system_planet = {
							limit = {
								is_asteroid = yes
							}
							remove_planet = yes
						}
						every_system_planet = {
							limit = {
								colonizable_planet = no
								is_star = no
								is_asteroid = no
								NOT = { is_planet_class = pc_gas_giant }
							}
							change_pc = pc_broken
						}
						every_system_planet = {
							limit = {
								colonizable_planet = no
								is_star = no
								is_planet_class = pc_gas_giant
							}
							change_pc = pc_molten
							add_modifier = { modifier = chthonian_planet }
							every_moon = {
								limit = {
									colonizable_planet = no
								}
								remove_planet = yes
							}
							while = {
								limit = {
									planet_size >= 20
								}
								change_planet_size = -6
							}
						}
					}
				}
			}
		}
	}
	
	ai_weight = {
		weight = 0
	}
}
