#Lithoid origin
# Buried Lithoids are generated on lithoid-origin colonisation
d_buried_lithoids = {
	icon = d_buried_lithoids
	time = 360

	is_for_colonizable = yes
	category = deposit_cat_rare_blocker

	potential = { always = no }

	on_cleared = {
		if = {
			limit = {
				NOT = { owner = { OR = {
					has_country_flag = decision_lithoid
					has_origin = origin_lithoid
				} } }
			}
			owner = { add_resource = { minerals = 300 } }
		}

		if = {
			limit = {
				owner = { 	OR = {
					has_country_flag = decision_lithoid
					has_origin = origin_lithoid
				} }
			}
			create_pop_group = {
				species = owner
			}
		}
	}

	resources = {
		category = deposit_blockers
		cost = {
			trigger = {
				NOT = { owner = { 	OR = {
					has_country_flag = decision_lithoid
					has_origin = origin_lithoid
				} } }
			}
			energy = 300
		}
		cost = {
			trigger = {
				owner = { 	OR = {
					has_country_flag = decision_lithoid
					has_origin = origin_lithoid
				} }
			}
			minerals = 1000
		}
	}

	ai_weight = {
		weight = 2

		modifier = {
			factor = 0
			AND = {
				owner = { 	OR = {
					has_country_flag = decision_lithoid
					has_origin = origin_lithoid
				} }
				from = { num_unemployed > 0 }
			}
		}
	}
}
