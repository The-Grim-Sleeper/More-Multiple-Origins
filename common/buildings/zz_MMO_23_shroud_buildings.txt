	can_be_ruined = no
	can_be_disabled = no

		urban
	}

		exists = owner

				has_origin = origin_shroud_forged

	}

	empire_limit = {
		base = 1
		modifier = {
			add = materiality_engine_available
			check_variable = {
				which = materiality_engine_available
				value > 0
			}
		}
	}

	}

	resources = {
		category = planet_buildings

	}

	triggered_desc = {
		text = physics_unit_is_extradimensional_research_unit_tt
	}

				add_trust = {
					who = root.owner
					amount = @mindwarden_opinion_lost_materiality_engine
				}
			}
		}
	}
