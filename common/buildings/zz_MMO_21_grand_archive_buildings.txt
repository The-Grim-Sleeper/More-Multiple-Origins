
wildlife_ranch = {
	base_buildtime = @b1_time
	planet_limit = 1
	category = amenity

	building_sets = {
		government
		urban
	}

	potential = {
		owner = {
			OR = {
				has_origin = origin_primal_calling
				has_country_flag = decision_primal_calling
			}
		}
		NOR = {
			has_building = hunting_grounds
			has_building = wildlife_sanctuary
			has_building = primal_arena
			has_building = symbiosis_nexus
		}
	}

	destroy_trigger = {
		owner = {
			OR = {
				NOR = { has_origin = origin_primal_calling has_country_flag = decision_primal_calling }
				wrangler_is_trophy_hunter_trigger = yes
				wrangler_is_animal_handler_trigger = yes
				wrangler_is_beastfighter_trigger = yes
			}
		}
	}

	convert_to = {
		hunting_grounds
		wildlife_sanctuary
		primal_arena
		symbiosis_nexus
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = @b1_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		upkeep = {
			energy = @b1_upkeep
		}
	}

	inline_script = {
		script = jobs/wranglers_add
		AMOUNT = @building_static_jobs
	}

	upgrades = {
		"wildlife_ranch_2"
	}
}
