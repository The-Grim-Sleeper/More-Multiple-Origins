
@festivalDuration = 1800 #days

namespace = leviathans
# Opus Sentinel (Unplugged)
country_event = {
    id = leviathans.559
    base = leviathans.500
    desc_clear = yes
    option_clear = yes

    title = leviathans.559.name
    desc = {
        text = leviathans.559.desc
        trigger = {
            NOT = { has_unplugged_origin = yes }
        }
    }
    desc = {
        text = leviathans.559.desc.unplugged
        trigger = {
            has_unplugged_origin = yes
        }
    }

    # More info
    option = {
        name = leviathans.559.a
        response_text = leviathans.559.a.reply
        is_dialog_only = yes
    }
    # Unplugged response
    option = {
        name = leviathans.559.b
        response_text = leviathans.559.b.reply
        is_dialog_only = yes
        trigger = {
            has_unplugged_origin = yes
            NOT = { has_country_flag = curator_unplugged_notalk }
        }
        hidden_effect = {
            set_country_flag = curator_unplugged_notalk
        }
    }
    # Purchase location
    option = {
        name = leviathans.559.c
        response_text = leviathans.559.c.reply
        trigger = {
            NOR = {
                has_country_flag = bought_opus_sentinel_info
                has_country_flag = opus_sentinel_encounter
            }
            OR = { # is able to accept relevant event chains
                NOT = { has_unplugged_origin = yes }
                AND = {
                    has_unplugged_origin = yes
                    has_event_chain = unplugged_origin_machine_legacy_chain
                }
            }
            exists = event_target:origin_unplugged_opus_sentinel
        }
        allow = {
            event_target:curator_enclave_country = {
                trust = {
                    who = root
                    value >= 40
                }
            }
            resource_stockpile_compare = {
                resource = energy
                value >= 1000
            }
            nomad_can_afford_enclave_cost = { COST = 1000 }
        }
        add_resource = {
            energy = -1000
        }
        custom_tooltip = opinion20
        hidden_effect = {
            set_country_flag = bought_opus_sentinel_info
            event_target:curator_enclave_country = {
                add_trust = {
                    who = root
                    amount = 20
                }
            }
            # Unplugged: set a PoI (inside Machine Legacy event chain)
            if = {
                limit = {
                    has_unplugged_origin = yes
                }
                create_point_of_interest = {
                    id = unplugged_curator_poi_opus_sentinel
                    name = curator_poi_title
                    desc = unplugged_curator_poi_opus_sentinel_desc
                    event_chain = unplugged_origin_machine_legacy_chain
                    location = event_target:origin_unplugged_machine_legacy_core
                }
            }
            # Non-Unplugged: set a PoI (custom version of 'create_curator_guardian_poi')
            else = {
                if = {
                    limit = {
                        NOT = { has_event_chain = curator_poi_chain }
                    }
                    begin_event_chain = {
                        event_chain = curator_poi_chain
                        target = event_target:origin_unplugged_opus_sentinel
                    }
                }
                create_point_of_interest = {
                    id = curator_poi_opus_sentinel
                    name = curator_poi_title
                    desc = unplugged_curator_poi_opus_sentinel_desc
                    event_chain = curator_poi_chain
                    location = event_target:origin_unplugged_machine_legacy_core
                }
            }
        }
    }

    # Standard options
    inline_script = enclaves/curators_guardians_options

    after = {
        hidden_effect = {
            remove_country_flag = curator_unplugged_notalk # used in Unplugged dialog
        }
    }
}
